namespace Sample.Application;

public sealed class Counter
{
    public int Total = 5;

    public int Step { get; set; }

    public void Run()
    {
        Total = 1;
        Total += Step;
        Total++;
        Bump(ref Total);
        Fill(out Total);
        var snapshot = Total;
        Step = snapshot;
        var seen = System.Threading.Volatile.Read(ref Total);
        System.Threading.Interlocked.Exchange(ref Total, seen);
    }

    private static void Bump(ref int value) => value++;

    private static void Fill(out int value) => value = 7;
}
